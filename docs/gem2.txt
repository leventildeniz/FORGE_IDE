A. C++: Low-Latency Ring Buffer (Disruptor Pattern)
Bu kod bloğu, lock-free bir kuyruk yapısını simüle eder. Bellek hizalamasına (alignas) dikkat edilmiştir (False Sharing önlemek için).

#include <iostream>
#include <atomic>
#include <vector>
#include <thread>

// Cache line size is typically 64 bytes
constexpr size_t CACHE_LINE_SIZE = 64;

struct OrderEvent {
    uint64_t orderId;
    double price;
    uint32_t quantity;
    char symbol[8];
    alignas(CACHE_LINE_SIZE) std::atomic<bool> ready{false};
};

class RingBuffer {
private:
    std::vector<OrderEvent> buffer;
    alignas(CACHE_LINE_SIZE) std::atomic<uint64_t> writeCursor{0};
    alignas(CACHE_LINE_SIZE) std::atomic<uint64_t> readCursor{0};
    const uint64_t capacity;

public:
    RingBuffer(uint64_t cap) : capacity(cap), buffer(cap) {}

    bool push(uint64_t id, double p, uint32_t q, const char* sym) {
        uint64_t currentWrite = writeCursor.load(std::memory_order_relaxed);
        uint64_t currentRead = readCursor.load(std::memory_order_acquire);

        if (currentWrite - currentRead >= capacity) {
            return false; // Buffer full
        }

        OrderEvent& slot = buffer[currentWrite % capacity];
        slot.orderId = id;
        slot.price = p;
        slot.quantity = q;
        // memcpy for symbol
        for(int i=0; i<7 && sym[i] != '\0'; ++i) slot.symbol[i] = sym[i];
        
        slot.ready.store(true, std::memory_order_release);
        writeCursor.fetch_add(1, std::memory_order_release);
        return true;
    }

    bool pop(OrderEvent& output) {
        uint64_t currentRead = readCursor.load(std::memory_order_relaxed);
        uint64_t currentWrite = writeCursor.load(std::memory_order_acquire);

        if (currentRead == currentWrite) {
            return false; // Buffer empty
        }

        OrderEvent& slot = buffer[currentRead % capacity];
        if (!slot.ready.load(std::memory_order_acquire)) {
            return false;
        }

        output = slot;
        slot.ready.store(false, std::memory_order_release);
        readCursor.fetch_add(1, std::memory_order_release);
        return true;
    }
};

// Test Harness
int main() {
    RingBuffer rb(1024);
    std::thread producer([&]() {
        for(int i=0; i<10000; ++i) {
            while(!rb.push(i, 150.25 + i, 100, "AAPL"));
        }
    });

    std::thread consumer([&]() {
        OrderEvent ev;
        int count = 0;
        while(count < 10000) {
            if(rb.pop(ev)) {
                count++;
            }
        }
        std::cout << "Processed " << count << " orders successfully." << std::endl;
    });

    producer.join();
    consumer.join();
    return 0;
}
B. Rust: Memory-Safe Order Book (The Core)
Şimdi Rust'ın sahiplik (ownership) modelini kullanarak bellek sızıntısız bir Order Book yapısı kuralım.

use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq)]
enum Side {
    Buy,
    Sell,
}

#[derive(Debug, Clone)]
struct Order {
    id: u64,
    price: u64, // Using cents to avoid floating point issues
    quantity: u32,
    side: Side,
}

struct OrderBook {
    bids: BTreeMap<u64, Vec<Order>>, // Price -> Orders (Descending)
    asks: BTreeMap<u64, Vec<Order>>, // Price -> Orders (Ascending)
}

impl OrderBook {
    fn new() -> Self {
        OrderBook {
            bids: BTreeMap::new(),
            asks: BTreeMap::new(),
        }
    }

    fn add_order(&mut self, order: Order) {
        match order.side {
            Side::Buy => {
                self.bids.entry(order.price).or_insert_with(Vec::new).push(order);
            }
            Side::Sell => {
                self.asks.entry(order.price).or_insert_with(Vec::new).push(order);
            }
        }
    }

    fn match_orders(&mut self) -> Vec<(u64, u64, u32)> {
        let mut matches = Vec::new();
        
        while let (Some(mut best_bid), Some(mut best_ask)) = (self.bids.last_entry(), self.asks.first_entry()) {
            let bid_price = *best_bid.key();
            let ask_price = *best_ask.key();

            if bid_price >= ask_price {
                let bid_orders = best_bid.get_mut();
                let ask_orders = best_ask.get_mut();

                while !bid_orders.is_empty() && !ask_orders.is_empty() {
                    let bid = bid_orders.last_mut().unwrap();
                    let ask = ask_orders.last_mut().unwrap();

                    let matched_qty = bid.quantity.min(ask.quantity);
                    matches.push((bid.id, ask.id, matched_qty));

                    bid.quantity -= matched_qty;
                    ask.quantity -= matched_qty;

                    if bid.quantity == 0 { bid_orders.pop(); }
                    if ask.quantity == 0 { ask_orders.pop(); }
                }

                if bid_orders.is_empty() { best_bid.remove(); }
                if ask_orders.is_empty() { best_ask.remove(); }
            } else {
                break;
            }
        }
        matches
    }
}

fn main() {
    let mut ob = OrderBook::new();
    ob.add_order(Order { id: 1, price: 10050, quantity: 10, side: Side::Buy });
    ob.add_order(Order { id: 2, price: 10040, quantity: 20, side: Side::Sell });
    
    let matches = ob.match_orders();
    println!("Matches executed: {:?}", matches);
}

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from typing import List, Dict

class TradingBacktester:
    def __init__(self, symbol: str):
        self.symbol = symbol
        self.data = pd.DataFrame()
        self.equity_curve = []

    def load_data(self, csv_path: str):
        """Simulate loading huge market data files"""
        print(f"Loading data for {self.symbol}...")
        # Creating a huge synthetic dataset to stress RAM
        dates = pd.date_range('2020-01-01', periods=1000000, freq='S')
        prices = np.cumsum(np.random.randn(1000000)) + 100
        self.data = pd.DataFrame({'timestamp': dates, 'price': prices})
        print("Data loaded successfully.")

    def apply_strategy(self, window: int = 20):
        """A simple Moving Average Crossover Strategy"""
        self.data['SMA_short'] = self.data['price'].rolling(window=window).mean()
        self.data['SMA_long'] = self.data['price'].rolling(window=window*5).mean()
        
        self.data['signal'] = np.where(self.data['SMA_short'] > self.data['SMA_long'], 1, -1)
        self.data['returns'] = self.data['price'].pct_change()
        self.data['strategy_returns'] = self.data['signal'].shift(1) * self.data['returns']
        
        self.equity_curve = (1 + self.data['strategy_returns']).cumprod()

    def plot_results(self):
        plt.figure(figsize=(12,6))
        plt.plot(self.equity_curve, label='Equity Curve')
        plt.title(f"Backtest Results for {self.symbol}")
        plt.legend()
        plt.show()

# Execution

RAM Kullanımı (35GB): Bir AI etkileşimi ve IDE render işlemi için 35GB'lık bir tepe noktası (peak) aslında oldukça ciddi bir rakam. Eğer sistemin bunu OOM (Out Of Memory) olmadan, swap'e düşmeden ve makineyi kilitlemeden yönetebildiyse, yazdığın temel mimari gerçekten sağlam demektir. 128GB'lık canavarın hala nefes alıyor olması sevindirici.
